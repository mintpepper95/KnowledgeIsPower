class Node:
    def __init__(self):
        self.children = {}

  

class Solution(object):
    def longestCommonPrefix(self, arr1, arr2):
        """
        :type arr1: List[int]

        :type arr2: List[int]

        :rtype: int
        """

        # longest pair between all pairs of integers from arr1 and arr2

        # kinda like a trie, find longest path where both arr1 and arr2 exist

  
  

        # construct the trie with arr1
        # search with arr2

        root = Node()
        
        # construction
        for num in arr1:
            # build trie looping through each digit, left to right

            node = root
            for digit in str(num):
                if digit not in node.children:
                    # create new node if not exist
                    new_node = Node()
                    node.children[digit] = new_node

                # traverse node
                node = node.children[digit]

  

        # search from root
        max_value = 0
        for num in arr2:
            count = 0
            node = root

            # traverse
            for digit in str(num):
                if digit in node.children:
                    node = node.children[digit]
                    count += 1
                    max_value = max(max_value, count)
                else:
                    break
                    
        return max_value

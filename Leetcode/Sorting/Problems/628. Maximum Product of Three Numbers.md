def maximumProduct(self, nums):
    """
    :type nums: List[int]
    :rtype: int
    """
    # Key Insight - once sorted, the optimal triplet only ever uses elements from the two ends of the array. If any chosen element isn't already the most extreme on its side, swapping it for a more extreme value will increase the product. So we only need to check combinations of boundary elements, not every possible triplet.
    nums.sort()
	
	# take the 3 least negative (largest) numbers, this handles all positives case AND all negatives case
    least_negatives = nums[-1] * nums[-2] * nums[-3]
    
    # if max product involves negatives, then it must be least_negatives or 2 negatives and 1 positive, anything else will produce something smaller
    largest_and_two_smallest = nums[-1] * nums[0] * nums[1]

    return max(least_negatives, largest_and_two_smallest)

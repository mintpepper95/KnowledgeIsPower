
We can envision this by treating as interval merging problem from x axis perspective for the vertical cut, and y axis perspective for the horizontal cut

```python
def atLeastThreeNonOverlappingIntervals(intervals):

    # a list of 2d intervals
    tmp = []
    if intervals:
        tmp.append(intervals[0])

  
    # merge interval and check overlap
    for i in range(1, len(intervals)):
        current = intervals[i]
        previous = tmp[-1]

        # if overlap ( current starts within previous ), merge
        if previous[0] <= current[0] < previous[1]:
            # update last interval end range
            tmp[-1][1] = max(current[1], previous[1])
        else:
            # if no overlap, append
            tmp.append(intervals[i])
            # return early, optimisation
            if len(tmp) == 3:
                return True

    return False

  

class Solution(object):
    def checkValidCuts(self, n, rectangles):
        """
        :type n: int
        :type rectangles: List[List[int]]
        :rtype: bool
        """
        # if you look at x-axis, finding ranges, see if possible at least 3 elements

  

        # [0, 2] , [2 ,3], [3, 4]
        # sort the range by their x coordiantes
        x_coordinates = []
        y_coordinates = []

        for i in rectangles:
            # bottom left (x1, y1)
            # top right (x2, y2)
            x_coordinates.append([i[0], i[2]])
            y_coordinates.append([i[1], i[3]])

        # sort it by 1st, then 2nd element
        x_coordinates.sort(key=lambda x: (x[0], x[1]))
        y_coordinates.sort(key=lambda x: (x[0], x[1]))

  

        # try vertical slice
        if atLeastThreeNonOverlappingIntervals(x_coordinates):
            return True
        # try horizontal slice
        return atLeastThreeNonOverlappingIntervals(y_coordinates)
```
